import { Body, Controller, Header, Post } from '@nestjs/common';
import { z } from 'zod';
import { RequirePermission } from '../../common/auth/require-permission.decorator.js';
import { ZodBody } from '../../common/validation/zod-body.js';
import { requireTenantContext } from '../../common/context/index.js';
import { StaffOnboardingService } from './staff-onboarding.service.js';
import { assertScopeWithin } from './staff-scope.js';

const inviteSchema = z
  .object({
    email: z.string().min(3).max(254),
    roleKey: z.enum(['cashier', 'sales']),
    companyIds: z.array(z.string().uuid()).min(1).max(20),
    branchIds: z.array(z.string().uuid()).min(1).max(20),
  })
  .strict();

@Controller('org/staff-invites')
export class OwnerStaffInviteController {
  constructor(private readonly staff: StaffOnboardingService) {}

  @Post()
  @RequirePermission('staff:invite')
  @Header('Cache-Control', 'no-store')
  async invite(@Body(new ZodBody(inviteSchema)) dto: z.infer<typeof inviteSchema>) {
    const ctx = requireTenantContext(); // tenantId + userId + data scope from the SESSION only
    assertScopeWithin(
      { companyScope: ctx.companyScope, branchScope: ctx.branchScope },
      { companyIds: dto.companyIds, branchIds: dto.branchIds },
    );
    return this.staff.inviteStaff({
      tenantId: ctx.tenantId,
      ...dto,
      actor: { kind: 'TENANT_USER', id: ctx.userId },
    });
  }
}
